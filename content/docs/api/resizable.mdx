`platejs/resizable/react` exports two headless React components and their types.
The package owns pointer, touch, keyboard, RTL, ARIA, and transient resize
behavior. Registry components own styling and editor persistence.

## Installation

```bash
npm install platejs @tanstack/react-virtual
```

## Usage

```tsx
import type { PlateElementProps } from 'platejs/react';
import { PlateElement, useEditor, usePath } from 'platejs/react';
import { ImagePlugin } from 'platejs/media/react';
import { Resizable, ResizeHandle } from 'platejs/resizable/react';

export function ImageElement(props: PlateElementProps<typeof ImagePlugin>) {
  const editor = useEditor();
  const path = usePath();

  return (
    <PlateElement {...props}>
      <div contentEditable={false}>
        <Resizable
          align="center"
          minWidth={120}
          width={props.element.width}
          onResizeEnd={(width) =>
            editor.plugin(ImagePlugin).update.set({ width }, { at: path })
          }
        >
          <ResizeHandle direction="left" />
          <img
            alt={props.element.alt}
            src={props.element.url}
            style={{ width: '100%' }}
          />
          <ResizeHandle direction="right" />
        </Resizable>
      </div>
      {props.children}
    </PlateElement>
  );
}
```

`Resizable` owns the private family context consumed by nested handles. No
provider or raw resize store is part of the public API.

## `Resizable`

<API name="Resizable">
<APIOptions type="ResizableProps">
  <APIItem name="align" type="'center' | 'left' | 'right'" optional>
    Controls resize direction math. Defaults to `center`.
  </APIItem>
  <APIItem name="maxWidth" type="ResizeLength" optional>
    Maximum width. Defaults to `100%`.
  </APIItem>
  <APIItem name="minWidth" type="ResizeLength" optional>
    Minimum width. Defaults to `0`.
  </APIItem>
  <APIItem name="onResizeEnd" type="(width: ResizeLength) => void" optional>
    Commits the finished width to the owning feature.
  </APIItem>
  <APIItem name="width" type="ResizeLength" optional>
    Current width. Defaults to `100%`.
  </APIItem>
  <APIItem name="ref" type="React.Ref<HTMLDivElement>" optional>
    React 19 ref for the inner resizable element.
  </APIItem>
</APIOptions>
</API>

## `ResizeHandle`

<API name="ResizeHandle">
<APIOptions type="ResizeHandleProps">
  <APIItem name="direction" type="ResizeDirection" optional>
    Resize edge. Defaults to `left`.
  </APIItem>
  <APIItem name="initialSize" type="number" optional>
    Starting size. The handle reads its parent when omitted.
  </APIItem>
  <APIItem name="onHover" type="() => void" optional>
    Called during mouse or touch hover.
  </APIItem>
  <APIItem name="onHoverEnd" type="() => void" optional>
    Called when hover or resizing finishes.
  </APIItem>
  <APIItem name="onResize" type="(event: ResizeEvent) => void" optional>
    Overrides the nearest `Resizable` callback.
  </APIItem>
</APIOptions>
</API>

The handle renders a `slider`, disappears in read-only mode, and supports arrow
keys. Hold Shift for a larger keyboard step. Pointer gestures listen on the
handle's owning window. Pointer cancellation, window blur, unmount, read-only
changes, and external width changes discard the preview. ARIA values and bounds
use pixels; the value text preserves the configured unit.

## Types

| API | Contract |
| --- | --- |
| `ResizeDirection` | `'bottom'`, `'left'`, `'right'`, or `'top'`. |
| `ResizeLength` | `number` or `string`; pixel values such as `120` or `"120px"`, or percentages such as `"50%"`. |
| `ResizeEvent` | `{ delta; direction; finished; initialSize }` |
